import {test,expect} from '/private/tmp/s3-account-personalization/node_modules/@playwright/test';
import fs from 'node:fs';
const base=process.env.QA_BASE_URL!;
const fixtures=()=>JSON.parse(fs.readFileSync(process.env.QA_ACCOUNT_FIXTURES!,'utf8'));
const cookies=(name:'A'|'B')=>{const f=fixtures()[name];return Array.isArray(f)?f:f.cookies};
const empty={personalizationOn:false,subjects:[],lessOf:[],experiences:[],challenge:null};
async function bind(context:any,name:'A'|'B'){await context.clearCookies();await context.addCookies(cookies(name));}
async function me(context:any){const r=await context.request.get(base+'/api/account/me');expect(r.ok()).toBe(true);return r.json();}
async function getSettings(context:any){const r=await context.request.get(base+'/api/account/settings');expect(r.ok()).toBe(true);return r.json();}
async function put(context:any,patch:any){const identity=await me(context);const r=await context.request.put(base+'/api/account/settings',{headers:{origin:new URL(base).origin,'x-s3c-account':identity.tag},data:patch});expect(r.ok()).toBe(true);return r.json();}
async function instrument(page:any){
 const requests:any[]=[],errors:string[]=[];
 page.on('request',(r:any)=>{const url=new URL(r.url());const body=r.postData();let keys:string[]=[];if(body){try{keys=Object.keys(JSON.parse(body))}catch{keys=['nonjson']}}requests.push({path:url.pathname,query:url.search,method:r.method(),bodyKeys:keys,snapshotHeader:!!r.headers()['x-s3c-feed-snapshot']});});page.on('pageerror',(e:any)=>errors.push(e.message));
 await page.addInitScript(()=>{const writes:any[]=[];const set=Storage.prototype.setItem;Storage.prototype.setItem=function(key,value){let keys:string[]=[];try{keys=Object.keys(JSON.parse(value))}catch{}writes.push({key:String(key),keys});return set.call(this,key,value)};(window as any).__preferencePrivacy={writes};});
 return {requests,errors};
}
async function assertBoundaries(page:any,observed:any){const writes=await page.evaluate(()=>(window as any).__preferencePrivacy.writes);expect(writes.filter((w:any)=>/settings|preference/i.test(w.key)||w.keys.some((k:string)=>['experiences','challenge','personalizationOn','subjects','lessOf'].includes(k)))).toEqual([]);expect(observed.requests.filter((r:any)=>/[?&](snapshot|subjects|experiences|challenge|accountTag)=/i.test(r.query))).toEqual([]);expect(observed.requests.filter((r:any)=>r.path!=='/api/account/settings'&&r.bodyKeys.some((k:string)=>['experiences','challenge','personalizationOn','subjects','lessOf','snapshot','accountTag'].includes(k)))).toEqual([]);expect(observed.errors).toEqual([]);return writes;}
